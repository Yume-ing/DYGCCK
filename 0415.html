🚀 JavaScript 基础知识学习

1. 变量声明

// var: 函数作用域，可重复声明（不推荐）
var name = "Alice";
var name = "Bob"; // 可以

// let: 块级作用域，不可重复声明（推荐）
let age = 20;

// const: 常量，声明后不能修改（推荐）
const PI = 3.14159;

2. 数据类型

// 基本数据类型
let str = "Hello World";      // 字符串 String
let num = 42;                  // 数字 Number
let bool = true;               // 布尔 Boolean
let nothing = null;            // 空值 Null
let notDefined;                // 未定义 Undefined

// 引用数据类型
let arr = [1, 2, 3, 4, 5];     // 数组 Array
let obj = {name: "张三", age: 20};  // 对象 Object

3. 函数定义

// 函数声明
function greet(name) {
    return "你好，" + name + "！";
}

// 箭头函数（ES6）
const add = (a, b) => a + b;

// 默认参数
function sayHello(name = "访客") {
    return "欢迎，" + name;
}

4. 条件语句

let score = 85;

// if...else if...else
if (score >= 90) {
    console.log("优秀");
} else if (score >= 80) {
    console.log("良好");
} else if (score >= 60) {
    console.log("及格");
} else {
    console.log("不及格");
}

// 三元运算符
let result = score >= 60 ? "通过" : "未通过";

5. 循环语句

// for 循环
for (let i = 0; i < 5; i++) {
    console.log("第" + i + "次循环");
}

// while 循环
let count = 0;
while (count < 3) {
    console.log("while: " + count);
    count++;
}

// for...of 遍历数组
let fruits = ["苹果", "香蕉", "橙子"];
for (let fruit of fruits) {
    console.log(fruit);
}

6. 数组方法

let numbers = [1, 2, 3, 4, 5];

// 高阶函数
let doubled = numbers.map(n => n * 2);     // [2,4,6,8,10]
let evens = numbers.filter(n => n % 2 === 0); // 过滤偶数 [2,4]
let sum = numbers.reduce((a, b) => a + b, 0); // 求和 15

// 其他常用
numbers.includes(3);       // true
numbers.indexOf(4);        // 3

7. 对象操作

// 创建对象
let student = {
    name: "王五",
    age: 20,
    scores: [85, 90, 78],
    
    getAverage: function() {
        let total = this.scores.reduce((a, b) => a + b, 0);
        return total / this.scores.length;
    }
};

// 访问属性
student.name;              // "王五"
student["age"];            // 20

// 解构赋值
let {name, age} = student;

8. 异步编程

// Promise
const fetchData = () => {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve("数据加载成功！");
        }, 1000);
    });
};

// async/await
async function getData() {
    let result = await fetchData();
    console.log(result);
}

9. DOM 操作

// 获取元素
let element = document.getElementById("myId");

// 修改内容
element.innerHTML = "新内容";

// 修改样式
element.style.color = "red";

// 添加事件监听
element.addEventListener("click", function() {
    alert("被点击了！");
});
